01

03

/* 核心精细化焦点系统 (Champagne Dark v1.2) */
.leanback-card-focus {
  transition: transform 200ms cubic-bezier(0.16, 1, 0.3, 1), box-shadow 200ms cubic-bezier(0.16, 1, 0.3, 1);
  will-change: transform, box-shadow;
}
.leanback-card-focus:focus-visible, .leanback-card-focus.is-focused {
  transform: scale(1.03);
  z-index: 50;
  box-shadow: 0 0 0 1.5px rgba(252, 213, 139, 0.95), 0 0 16px -2px rgba(229, 169, 88, 0.55), 0 0 32px 4px rgba(245, 158, 11, 0.22);
}
/* 遥控器按下瞬态压缩 */
.leanback-card-focus:active {
  transform: scale(0.985);
  box-shadow: 0 0 0 2px #FFE8A3, 0 0 24px rgba(255, 232, 163, 0.8);
}
